# patch-reach - MAIN environment. Everything that touches a segmentor.
#
# This is the env the overwhelming majority of the thesis ran under: every
# attack, evaluation, defence and analysis script (scripts/train.py,
# overfit.py, overfit_population.py, evaluate.py, transfer_matrix.py,
# clean_baseline.py, measure_*.py, sweep_operating_point.py, iap_attack.py,
# defences/, analysis/). On the cluster it is
# $WS/miniconda3/envs/thesis_backup3 and the SLURM launchers call that
# interpreter by absolute path rather than `conda activate` (that conda install
# moved from /hkfs to /pfs and its shebangs still point at the old path).
#
# THE WHOLE STACK IS PINNED BY ONE DECISION: mmcv 1.x + mmseg 0.x.
# patchreach calls `from mmcv import Config` and `mmseg.models.build_segmentor`
# - the 1.x/0.x API, not the 2.x one - and the released checkpoints were
# trained under it. mmcv-full 1.5.0 is compiled against torch 1.11.0+cu113, so
# torch is pinned BY mmcv rather than the other way round. Treat
# torch / torchvision / mmcv-full / mmsegmentation / mmdet / mmcls as one
# atomic set: moving any one of them means rebuilding mmcv from source and
# revalidating every number in the results chapter, because mmcv/mmseg minor
# versions move the config schema and a config that builds a DIFFERENT variant
# from the checkpoint loads with missing-key warnings and still runs.
#
# A SINGLE `pip install -r` IS NOT ENOUGH. pycocotools compiles at install
# time, and mmcv-full compiles too whenever no wheel matches the resolved
# torch, so numpy, setuptools and ninja have to be importable BEFORE the
# resolver gets that far. The order below is not cosmetic:
#
#   conda env create -f envs/main.yml
#   conda activate patchreach
#   pip install numpy==1.26.4 setuptools==60.2.0 ninja==1.13.0
#   pip install -r envs/requirements-main.txt
#   pip install -e . --no-deps          # patchreach itself; deps are above
#
# STILL MISSING AFTERWARDS, both on purpose:
#   dcnv3        compiled from InternImage source, see the bottom of this file
#   checkpoints  `mim download mmsegmentation --config <name> --dest ...`
# SegFormer (b0-b5), SETR, Swin/UPerNet, PSPNet and the DeepLabs all run
# without dcnv3; only --arch internimage_* needs it.

# NEITHER INDEX IS OPTIONAL. cu113 torch is not on PyPI and mmcv-full wheels
# are not either. The mmcv index is per (CUDA, torch) pair, so these two lines
# are a matched pair: changing one invalidates the other.
--extra-index-url https://download.pytorch.org/whl/cu113
--find-links https://download.openmmlab.com/mmcv/dist/cu113/torch1.11.0/index.html
